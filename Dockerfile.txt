FROM node:18-slim

# Instalar FFmpeg, Python y utilidades del sistema para Linux
RUN apt-get update && apt-get install -y \
    ffmpeg \
    python3 \
    curl \
    git \
    && rm -rf /var/lib/apt/lists/*

# Instalar la versión ejecutable nativa de yt-dlp para Linux
RUN curl -L https://github.com/yt-dlp/yt-dlp/releases/latest/download/yt-dlp -o /usr/local/bin/yt-dlp \
    && chmod a+rx /usr/local/bin/yt-dlp

WORKDIR /app

COPY package*.json ./
RUN npm install

COPY . .

CMD ["node", "index.js"]
